"""Apply only the approved 400-set reporting exception and separate coverage-8 sensitivity."""
import ast
import hashlib
import importlib
import json
import os
from pathlib import Path
import sys
import time

import numpy as np

R=Path(__file__).resolve().parent
T=Path('/media/carruthers/mnt2/spacr/codex-worktrees/spacr-codex-work-1012fix')
sys.path[:0]=[str(R),str(T),str(R.parent/'1012')]


def extend_random_set_owner(base):
    """Extend the one analysis helper after the original calculation has finished, preserving its source."""
    path=R/'primary_fallback_readout.py'
    snapshot=R/'primary_fallback_readout.executed-v1.py'
    original=snapshot.read_text() if snapshot.exists() else path.read_text()
    assert hashlib.sha256(original.encode()).hexdigest()==base['script_sha256']
    if not snapshot.exists():
        snapshot.write_text(original)
    changed=original.replace('def random_sets(templates,coverage_counts,eligible):',
        'def random_sets(templates,coverage_counts,eligible,*,sets=500):')
    changed=changed.replace('    """Draw exactly five hundred sets per stratum by cycling its five coverage-count templates."""',
        '    """Draw the requested count by cycling coverage-count templates without replacement within a set."""')
    changed=changed.replace('    assert len(templates)==5\n    for i in range(N_SETS):\n        template=templates[i%5]\n',
        '    assert len(templates)>0\n    for i in range(int(sets)):\n        template=templates[i%len(templates)]\n')
    assert changed!=original
    compile(changed,str(path),'exec')
    old_node=next(n for n in ast.parse(original).body if isinstance(n,ast.FunctionDef) and n.name=='random_sets')
    namespace={'np':np,'N_SETS':500}
    exec(compile(ast.Module(body=[old_node],type_ignores=[]),str(snapshot),'exec'),namespace)
    assert path.read_text() in (original,changed)
    if path.read_text()!=changed:
        path.write_text(changed)
    return namespace['random_sets']


def main():
    """Reuse exact persisted p/view arrays; never recompute fallback scores or alter primary PASS inputs."""
    assert os.environ['CUDA_VISIBLE_DEVICES']==''
    start=time.monotonic()
    base=json.loads((R/'primary-fallback-readout.json').read_text())
    old_draw=extend_random_set_owner(base)
    P=importlib.import_module('primary_fallback_readout')
    hashes=P.input_hashes()
    assert hashes==base['input_sha256']
    _,names,views,_,coverage,planted,_,_,_=P.load_inputs()
    counts=coverage.sum(axis=1)
    index={gene:i for i,gene in enumerate(names)}
    eligible=~np.isin(names,planted.gene.astype(str))
    groups=[]
    for name,group in planted.groupby('complex',sort=False):
        groups.append({'complex':str(name),'view':str(group.view.iloc[0]),
                       'strength':float(group.strength.iloc[0]),
                       'members':np.array([index[str(g)] for g in group.gene],int)})
    first=[g['members'] for g in groups if g['view']==groups[0]['view'] and g['strength']==groups[0]['strength']]
    assert len(first)==5
    assert np.array_equal(np.stack(old_draw(first,counts,eligible)),np.stack(P.random_sets(first,counts,eligible)))
    owner_proof={'default500_matched_indices_byte_equal':True,'old_script_sha256':base['script_sha256'],
                 'extended_script_sha256':hashlib.sha256((R/'primary_fallback_readout.py').read_bytes()).hexdigest()}
    selected=[g for g in groups if g['view']=='conservation' and g['strength']==1]
    bad=[g for g in selected if np.any(counts[g['members']]==7)]
    feasible=[g for g in selected if g['complex'] not in {item['complex'] for item in bad}]
    assert len(selected)==5 and len(bad)==1 and bad[0]['complex']=='plant_conservation_s1_0' and len(feasible)==4
    members7=bad[0]['members'][counts[bad[0]['members']]==7]
    assert len(members7)==1 and np.sum(eligible&(counts==7))==0 and np.sum(eligible&(counts==8))==133
    out=R/'primary-arrays'
    metadata=json.loads((out/'planted-metadata.json').read_text())
    for suffix in ('p','view'):
        assert hashlib.sha256((out/('planted-'+suffix+'.npy')).read_bytes()).hexdigest()==metadata[suffix+'_file_sha256']
    p=np.load(out/'planted-p.npy',mmap_mode='r',allow_pickle=False)
    best=np.load(out/'planted-view.npy',mmap_mode='r',allow_pickle=False)
    target=next(i for i,view in enumerate(views) if view['name']=='conservation')
    assert base['null_details']['planted'][target]['view_index']==target
    templates=[g['members'] for g in feasible]
    controls=P.random_sets(templates,counts,eligible,sets=400)
    assert all(np.array_equal(counts[m],counts[templates[i%4]]) and eligible[m].all() and len(set(m))==8 for i,m in enumerate(controls))
    metrics=[P.set_metrics(p,best,m,target,len(names)) for m in controls]
    summary={key:P.summary([row[key] for row in metrics]) for key in ('pair_share','pair_p05_share_all_views','map')}
    row=next(row for row in base['strata'] if row['view']=='conservation' and row['strength']==1)
    assert row['random'] is None and row['random_sets']==0
    row.update(random_sets=400,random_sets_match_coverage_counts=True,random=summary,
               random_set_indices_sha256=hashlib.sha256(np.stack(controls).tobytes()).hexdigest(),
               reporting_exception='1 of 5 templates in conservation 1x has no coverage-matched null (its coverage-7 member is the only coverage-7 gene and is planted); 400 random sets instead of 500.')
    sensitivity_counts=counts.copy();sensitivity_counts[members7]=8
    sensitivity_controls=P.random_sets([bad[0]['members']],sensitivity_counts,eligible,sets=100)
    assert all(np.array_equal(counts[m],sensitivity_counts[bad[0]['members']]) and eligible[m].all() and len(set(m))==8 for m in sensitivity_controls)
    sensitivity_metrics=[P.set_metrics(p,best,m,target,len(names)) for m in sensitivity_controls]
    observed=next(row for row in base['complexes'] if row['complex']==bad[0]['complex'])
    sensitivity={'label':'SENSITIVITY ONLY; excluded from PASS','complex':bad[0]['complex'],
                 'change':'Only the matching coverage count for the coverage-7 member is8, using the nearest133-gene eligible pool; no measurement, profile, p value or planted template changes.',
                 'random_sets':100,'seed':0,'observed_planted':observed['planted'],'observed_no_plant':observed['no_plant'],
                 'random':{key:P.summary([item[key] for item in sensitivity_metrics]) for key in ('pair_share','pair_p05_share_all_views','map')},
                 'random_set_indices_sha256':hashlib.sha256(np.stack(sensitivity_controls).tobytes()).hexdigest()}
    base['control_blockers_resolved_by_reporting']=base['control_blockers']
    base['control_blockers']=[];base['readout_complete']=True
    base['reporting_exception_approval_inbox']='549a3215fe7c5c95740be15160fa122d16a9684eae1f22768c664a61e70d51c5'
    base['conservation1_sensitivity']=sensitivity
    base['supplement']={'script_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
                        'owner_extension_proof':owner_proof,'seconds':time.monotonic()-start}
    assert P.input_hashes()==hashes
    original=json.loads((R/'primary-fallback-readout.json').read_text())
    for key in ('gates','pooled8','trend','no_plant_p05_share_all260_complexes','all_reported_gates_pass'):
        assert base[key]==original[key]
    (R/'primary-fallback-complete.json').write_text(json.dumps(base,indent=2)+'\n')
    (R/'conservation1-control-supplement.json').write_text(json.dumps({'main400':summary,'sensitivity_only':sensitivity,'owner_extension_proof':owner_proof},indent=2)+'\n')
    print('SUPPLEMENT_COMPLETE',json.dumps({'main400':summary,'sensitivity_only':sensitivity,'unchanged_gates':base['gates']}),flush=True)


if __name__=='__main__':
    main()
