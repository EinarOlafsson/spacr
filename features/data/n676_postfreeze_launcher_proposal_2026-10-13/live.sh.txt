#!/usr/bin/env bash
# make_host_jobs --launcher contract: STAGE [run_native options] -- [capture args]
set -euo pipefail
: "${SPACR_FINAL_CAPTURE_REPO:?Supply the approved final isolated worktree}"
: "${SPACR_FINAL_CAPTURE_REVISION:?Supply its exact final 40-character clean commit}"
: "${SPACR_CAPTURE_PYTHON:?Supply the existing app Python executable}"
[[ $# -ge 1 ]] || { echo 'usage: live.sh STAGE [run_native options] -- [capture args]' >&2; exit 2; }
launcher_dir=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)
capture_stage=$1
shift
cd -- "$SPACR_FINAL_CAPTURE_REPO"
exec "$SPACR_FINAL_CAPTURE_REPO/tools/run_capped.sh" 12G \
    "$SPACR_CAPTURE_PYTHON" "$launcher_dir/bootstrap.py" \
    --repo "$SPACR_FINAL_CAPTURE_REPO" --revision "$SPACR_FINAL_CAPTURE_REVISION" \
    --stage "$capture_stage" -- "$@"
