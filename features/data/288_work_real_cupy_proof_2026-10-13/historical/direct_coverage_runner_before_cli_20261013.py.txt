def require(condition, message):
    if not condition:
        raise RuntimeError(message)


def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b''):
            digest.update(chunk)
    return digest.hexdigest()


def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def command(*args):
    return subprocess.check_output(args, cwd=ROOT, text=True,
                                   stderr=subprocess.STDOUT, timeout=15)


def distribution_module_file(distribution, relative):
    path = Path(importlib.metadata.distribution(distribution).locate_file(relative)).resolve()
    require(path.is_file(), 'Installed distribution module is absent: ' + str(path))
    return path


def validate_sources(request):
    require(command('git', 'rev-parse', 'HEAD').strip() == SOURCE, 'Wrong source HEAD')
    require(command('git', 'status', '--porcelain') == '', 'Worktree is not clean')
    bindings = {}
    for row in request['files']:
        path = ROOT / row['path']
        require(path.stat().st_size == row['bytes'] and sha(path) == row['sha256'],
                'Source/test/tool hash changed: ' + row['path'])
        bindings[row['path']] = {'sha256': row['sha256'], 'bytes': row['bytes']}
    return bindings


def lease_evidence():
    """Check the actual scheduler holder and its locked descriptor before CUDA."""
    for name, default in [('GPU_IDLE_S', '360'), ('GPU_GAP_S', '600'), ('GPU_POLL_S', '15')]:
        require(os.environ.get(name, default) == default, 'Scheduler guard override: ' + name)
    folder = Path.home() / '.spacr/gpu'
    holder = folder / 'holder'
    raw = holder.read_text().strip()
    parts = dict(piece.split('=', 1) for piece in raw.split())
    require(parts.get('label') == LABEL and parts.get('checkout') == str(ROOT),
            'No matching scheduler holder for this worktree')
    pid = int(parts['pid'])
    require(pid != os.getpid(), 'Wrapper cannot be its own scheduler holder')
    os.kill(pid, 0)
    ancestry, seen = [], set()
    ancestor = os.getpid()
    for _ in range(128):
        require(ancestor > 0 and ancestor not in seen, 'Invalid scheduler ancestry chain')
        seen.add(ancestor)
        status = (Path('/proc') / str(ancestor) / 'status').read_text().splitlines()
        parent = int(next(line.split(':', 1)[1] for line in status if line.startswith('PPid:')))
        ancestry.append({'pid': ancestor, 'parent_pid': parent})
        if ancestor == pid:
            break
        require(parent > 0, 'Collector is not descended from the matching scheduler holder')
        ancestor = parent
    else:
        raise RuntimeError('Scheduler ancestry exceeds the bounded traversal')
    cmdline = (Path('/proc') / str(pid) / 'cmdline').read_bytes().split(b'\0')
    arguments = [item.decode() for item in cmdline if item]
    for token in [str(ROOT / 'tools/gpu_turn.sh'), LABEL, 'timeout',
                  '--signal=TERM', '--kill-after=15s', '180s',
                  str(ROOT / 'tools/run_capped.sh'), '6G', PYTHON, str(Path(__file__).resolve())]:
        require(token in arguments, 'Scheduler command lacks required token: ' + token)
    lock = folder / 'lock'
    locked_fd = Path('/proc') / str(pid) / 'fd/9'
    require(locked_fd.stat().st_ino == lock.stat().st_ino
            and locked_fd.stat().st_dev == lock.stat().st_dev,
            'Scheduler descriptor does not identify the shared GPU lock')
    with lock.open('r+') as handle:
        try:
            fcntl.flock(handle, fcntl.LOCK_EX | fcntl.LOCK_NB)
        except BlockingIOError:
            pass
        else:
            fcntl.flock(handle, fcntl.LOCK_UN)
            raise RuntimeError('GPU scheduler lock is not held')
    entries = Path('/proc/self/cgroup').read_text().splitlines()
    cgroup = next(line.split(':', 2)[2] for line in entries if line.startswith('0::'))
    memory_max = Path('/sys/fs/cgroup') / cgroup.lstrip('/') / 'memory.max'
    limit = memory_max.read_text().strip()
    require(limit != 'max' and int(limit) <= 6 * 1024 ** 3, 'Missing host memory cap <=6G')
    return {'holder': raw, 'holder_sha256': sha(holder), 'holder_pid': pid,
            'scheduler_command': arguments, 'collector_to_holder_ancestry': ancestry,
            'memory_max': limit,
            'memory_max_path': str(memory_max), 'idle_seconds': 360, 'gap_seconds': 600}


def gpu_snapshot():
    """Only called after the scheduler lock/holder/cap checks succeeded."""
    raw = command('nvidia-smi', '--query-gpu=index,uuid,name,pci.bus_id,driver_version,memory.total,memory.free',
                  '--format=csv,noheader,nounits')
    fields = ['index', 'uuid', 'name', 'pci_bus_id', 'driver_version',
              'memory_total_MiB', 'memory_free_MiB']
    devices = [dict(zip(fields, [value.strip() for value in row]))
               for row in csv.reader(raw.splitlines())]
    visible = os.environ.get('CUDA_VISIBLE_DEVICES')
    selected = [row for row in devices if visible in (row['index'], row['uuid'])]
    require(len(selected) == 1, 'Select exactly one physical device by index or UUID')
    require(int(selected[0]['memory_free_MiB']) * 1024 ** 2 >= HEADROOM,
            'Insufficient >=1500MiB free device memory')
    processes = command('nvidia-smi', '--query-compute-apps=pid,gpu_uuid,process_name',
                        '--format=csv,noheader')
    pids = [row[0].strip() for row in csv.reader(processes.splitlines()) if row]
    require(all(pid == str(os.getpid()) for pid in pids),
            'Another compute process is present; do not disturb it')
    return {'selected': selected[0], 'all_devices': devices,
            'compute_processes_csv': processes, 'CUDA_VISIBLE_DEVICES': visible}


def validate_junit(path):
    root = ET.parse(path).getroot()
    cases = list(root.iter('testcase'))
    expected = {node.split('::')[1] for node in NODEIDS}
    require(len(cases) == 3 and {case.get('name') for case in cases} == expected,
            'JUnit does not identify exactly the three requested CuPy cases')
    require(all(case.get('classname', '').endswith('test_the_ops_primitives_agree_on_every_backend')
                for case in cases), 'JUnit case from unexpected test module')
    require(not any(list(root.iter(tag)) for tag in ['skipped', 'failure', 'error']),
            'JUnit contains a skip/failure/error')
    suites = [root] if root.tag == 'testsuite' else list(root.iter('testsuite'))
    require(sum(int(s.get('tests', 0)) for s in suites) == 3
            and all(int(s.get(key, 0)) == 0 for s in suites
                    for key in ['skipped', 'failures', 'errors']), 'JUnit totals differ')
    return {'tests': 3, 'passed': 3, 'skipped': 0, 'failed': 0, 'errors': 0,
            'cases': [{'name': c.get('name'), 'classname': c.get('classname')} for c in cases]}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--approval', type=Path, required=True)
    parser.add_argument('--output', type=Path, required=True)
    args = parser.parse_args()
    approval = json.loads(args.approval.read_text())
    proposal = SCRATCH / 'direct-coverage-716-proposal.json'
    require(approval.get('schema') == 1 and approval.get('approved') is True
            and approval.get('approved_by') == 'Home'
            and approval.get('source_candidate') == SOURCE
            and approval.get('coverage_version') == '7.16.0'
            and approval.get('collector') == 'Coverage API'
            and approval.get('wrapper_sha256') == sha(__file__)
            and approval.get('proposal_sha256') == sha(proposal),
            'Missing actual Home approval bound to this wrapper/proposal/version')
    require(args.output.resolve().is_relative_to(SCRATCH), 'Output must stay in authorized scratch')
    args.output.mkdir(parents=True, exist_ok=False)
    output = args.output.resolve()
    receipt = {'schema': 1, 'started_at': now(), 'source_candidate': SOURCE,
               'wrapper_sha256': sha(__file__), 'proposal_sha256': sha(proposal),
               'approval_sha256': sha(args.approval), 'pid': os.getpid(),
               'hosted_CI_evidence': False, 'coverage_union_measured': False,
               'accepted': False, 'inner_checks_passed': False,
               'actual_process_exit_code': None,
               'removed_editable_finders': REMOVED_EDITABLE_FINDERS}
    status = 1
    try:
        os.chdir(ROOT)
        require(sys.executable == PYTHON, 'Unexpected Python executable')
        require(platform.python_version() == '3.13.14', 'Unexpected Python version')
        require(sys.path[0] == str(ROOT), 'Worktree must be first on sys.path')
        for name, expected in [('QT_QPA_PLATFORM', 'offscreen'), ('PYTEST_QT_API', 'pyside6'),
                               ('SPACR_NO_GL', '1'), ('OMP_NUM_THREADS', '1'),
                               ('MKL_NUM_THREADS', '1'), ('OPENBLAS_NUM_THREADS', '1')]:
            require(os.environ.get(name) == expected, 'Incorrect process environment: ' + name)
        require(os.environ.get('COVERAGE_FILE') == str(output / '.coverage-f288-real-cupy'),
                'Subprocess coverage file must identify this private output')
        spec = importlib.util.find_spec('spacr')
        require(spec is not None and Path(spec.origin).resolve().parent == ROOT / 'spacr',
                'spaCR resolves outside the pinned worktree')
        require('spacr.ops_accel' not in sys.modules, 'Module imported before coverage')
        require('cupy' not in sys.modules and 'torch' not in sys.modules,
                'CUDA-capable libraries were imported before the lease checks')
        proposal_data = json.loads(proposal.read_text())
        request_path = SCRATCH / 'requested-proposal.json'
        require(sha(request_path) == proposal_data['evidence_sha256'][str(request_path)],
                'Home request changed after proposal review')
        request = json.loads(request_path.read_text())
        require(request['source_candidate'] == SOURCE and request['minimum_real_GPU_nodeids'] == NODEIDS,
                'Changed requested source or test inventory')
        receipt['source_bindings_before'] = validate_sources(request)
        receipt['environment'] = {'python': sys.version, 'executable': sys.executable,
                                  'platform': platform.platform(), 'packages': {}}
        for name, expected in PACKAGES.items():
            actual = importlib.metadata.version(name)
            require(actual == expected, 'Unexpected installed version: ' + name)
            receipt['environment']['packages'][name] = actual
        receipt['lease'] = lease_evidence()
        receipt['GPU_before_context'] = gpu_snapshot()
        # The first CUDA-capable import occurs only after the holder and lock checks.
        cupy_file = distribution_module_file('cupy-cuda12x', 'cupy/__init__.py')
        cupy_spec = importlib.util.find_spec('cupy')
        require(cupy_spec is not None and Path(cupy_spec.origin).resolve() == cupy_file,
                'CuPy resolves outside its installed distribution')
        import cupy as cp
        require(Path(cp.__file__).resolve() == cupy_file,
                'Actual CuPy module is outside its installed distribution')
        require(cp.__version__ == PACKAGES['cupy-cuda12x'], 'Actual CuPy version differs from metadata')
        require(cp.cuda.runtime.getDeviceCount() == 1, 'Expected one visible CUDA device')
        cp.cuda.Device(0).use()
        free, total = cp.cuda.runtime.memGetInfo()
        require(free >= HEADROOM, 'Context initialization leaves <1500MiB headroom')
        properties = cp.cuda.runtime.getDeviceProperties(0)
        pci_bus_id = cp.cuda.Device(0).pci_bus_id
        def bus_identity(value):
            domain, bus, slot = value.lower().split(':')
            device, function = slot.split('.')
            return tuple(int(component, 16) for component in (domain, bus, device, function))
        require(bus_identity(pci_bus_id) == bus_identity(receipt['GPU_before_context']['selected']['pci_bus_id']),
                'CuPy initialized a different physical device')
        configuration_text = io.StringIO()
        with contextlib.redirect_stdout(configuration_text):
            cp.show_config(_full=True)
        receipt['CuPy'] = {'version': cp.__version__, 'path': cp.__file__,
                           'distribution_module_file': str(cupy_file),
                           'pci_bus_id': pci_bus_id,
                           'runtime_version': cp.cuda.runtime.runtimeGetVersion(),
                           'driver_version': cp.cuda.runtime.driverGetVersion(),
                           'device_properties': {str(k): ({'hex': v.hex(), 'text': v.decode(errors='replace')}
                                                         if isinstance(v, bytes) else v)
                                                 for k, v in properties.items()},
                           'free_bytes_after_context': free, 'total_bytes': total,
                           'configuration': configuration_text.getvalue()}
        coverage_file = distribution_module_file('coverage', 'coverage/__init__.py')
        coverage_spec = importlib.util.find_spec('coverage')
        require(coverage_spec is not None and Path(coverage_spec.origin).resolve() == coverage_file,
                'Coverage resolves outside its installed distribution')
        import coverage
        require(Path(coverage.__file__).resolve() == coverage_file,
                'Actual coverage module is outside its installed distribution')
        require(coverage.__version__ == PACKAGES['coverage'], 'Actual coverage version differs from metadata')
        raw_db = output / '.coverage-f288-real-cupy'
        cov = coverage.Coverage(data_file=str(raw_db), data_suffix=False, branch=True,
                                source=['spacr.ops_accel'], config_file=str(ROOT / '.coveragerc'),
                                context='work-f288-real-cupy')
        configuration = {key: cov.get_option(key) for key in
                         ['run:branch', 'run:source', 'run:parallel', 'run:relative_files',
                          'run:patch', 'run:sigterm', 'report:exclude_lines', 'report:partial_branches']}
        require(configuration['run:branch'] is True
                and configuration['run:source'] == ['spacr.ops_accel']
                and configuration['run:relative_files'] is True
                and configuration['report:exclude_lines'] == []
                and configuration['report:partial_branches'] == [],
                'Coverage would weaken the recorded denominator or branch defaults')
        receipt['coverage'] = {'version': coverage.__version__, 'path': coverage.__file__,
                               'distribution_module_file': str(coverage_file),
                               'effective_configuration': configuration,
                               'explicit_overrides': {'source': ['spacr.ops_accel'], 'data_suffix': False,
                                                      'context': 'work-f288-real-cupy', 'data_file': str(raw_db)}}
        xml = output / 'f288-real-cupy.xml'
        coverage_json = output / 'f288-real-cupy-coverage.json'
        cov.start()
        try:
            import pytest
            class HeadroomChecks:
                def pytest_runtest_setup(self, item):
                    require(item.nodeid in NODEIDS, 'Unexpected test collected')
                    lease_evidence()
                    free, _ = cp.cuda.runtime.memGetInfo()
                    require(free >= HEADROOM, 'Test begins with <1500MiB free GPU memory')
            arguments = ['-n', '0', *NODEIDS, '--junitxml=' + str(xml), '-rA']
            receipt['pytest_arguments'] = arguments
            receipt['pytest_exit_code'] = int(pytest.main(arguments, plugins=[HeadroomChecks()]))
        finally:
            cov.stop()
            cov.save()
            cov.json_report(morfs=[str(ROOT / 'spacr/ops_accel.py')], outfile=str(coverage_json),
                            pretty_print=True)
        require(receipt['pytest_exit_code'] == 0, 'Pytest did not exit successfully')
        actual_module = Path(sys.modules['spacr.ops_accel'].__file__).resolve()
        require(actual_module == ROOT / 'spacr/ops_accel.py', 'Tests imported a different module')
        receipt['actual_ops_module_path'] = str(actual_module)
        receipt['JUnit'] = validate_junit(xml)
        data = cov.get_data()
        require(data.has_arcs(), 'Coverage raw data has no branch arcs')
        measured = sorted(data.measured_files())
        require(len(measured) == 1 and (ROOT / measured[0]).resolve() == ROOT / 'spacr/ops_accel.py',
                'Coverage measured a different or unexpected source module')
        json_data = json.loads(coverage_json.read_text())
        require(json_data['meta']['version'] == '7.16.0'
                and json_data['meta']['branch_coverage'] is True, 'Unexpected coverage JSON version/scope')
        with sqlite3.connect('file:' + str(raw_db) + '?mode=ro', uri=True) as db:
            receipt['coverage']['DB_schema'] = db.execute('select version from coverage_schema').fetchone()[0]
            receipt['coverage']['DB_metadata'] = dict(db.execute('select key,value from meta'))
        receipt['coverage']['measured_files'] = measured
        receipt['coverage']['summary'] = json_data['files'][measured[0]]['summary']
        receipt['source_bindings_after'] = validate_sources(request)
        require(receipt['source_bindings_after'] == receipt['source_bindings_before'], 'Sources changed during tests')
        receipt['GPU_after_tests'] = gpu_snapshot()
        receipt['artifacts'] = {str(p): {'sha256': sha(p), 'bytes': p.stat().st_size}
                                for p in [raw_db, xml, coverage_json]}
        receipt['inner_checks_passed'] = True
        status = 0
    except BaseException as exc:
        receipt['failure'] = {'type': type(exc).__name__, 'message': str(exc), 'traceback': traceback.format_exc()}
        traceback.print_exc()
    finally:
        receipt['finished_at'] = now()
        receipt['intended_exit_code'] = status
        receipt['scope'] = 'Work real-device test proof only; outer scheduler terminal exit must be recorded separately'
        (output / 'execution-receipt.json').write_text(json.dumps(receipt, indent=2, default=str) + '\n')
    return status


if __name__ == '__main__':
    raise SystemExit(main())
